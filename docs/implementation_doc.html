import numpy as np

def pennes_bioheat_3d(T, Q_implant, dt=0.05, dx=1e-3):
    k = 0.52 # W/(m K)
    rho_c = 3.6e6 # J/(m3 K)
    w_b_cb = 0.02
    T_art = 37.0
    
    laplacian = (np.pad(T[1:,:,:], ((0,1),(0,0),(0,0))) + np.pad(T[:-1,:,:], ((1,0),(0,0),(0,0))) +
                 np.pad(T[:,1:,:], ((0,0),(0,1),(0,0))) + np.pad(T[:,:-1,:], ((0,0),(1,0),(0,0))) +
                 np.pad(T[:,:,1:], ((0,0),(0,0),(0,1))) + np.pad(T[:,:,:-1], ((0,0),(0,0),(1,0))) - 6*T) / (dx**2)
    
    dT_dt = (k * laplacian - w_b_cb * (T - T_art) + Q_implant) / rho_c
    return T + dT_dt * dt
